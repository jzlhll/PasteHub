cmake_minimum_required(VERSION 3.22)
project(PasteHubImage VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build configuration" FORCE)
endif()

set(webp_root "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libwebp")
file(GLOB webp_sources CONFIGURE_DEPENDS
    "${webp_root}/src/enc/*.c" "${webp_root}/src/dsp/*.c"
    "${webp_root}/src/utils/*.c" "${webp_root}/sharpyuv/*.c")
if(WIN32)
    file(GLOB webp_decoder_sources CONFIGURE_DEPENDS "${webp_root}/src/dec/*.c" "${webp_root}/src/demux/*.c")
    list(APPEND webp_sources ${webp_decoder_sources})
endif()
add_library(ph_webp OBJECT ${webp_sources})
target_include_directories(ph_webp PRIVATE "${webp_root}")
target_compile_definitions(ph_webp PRIVATE WEBP_USE_THREAD NDEBUG)

set(core_sources src/pastehub_image.cpp src/orientation.cpp)
if(ANDROID)
    list(APPEND core_sources platforms/android/AndroidBridge.cpp)
    set(library_type SHARED)
elseif(APPLE)
    list(APPEND core_sources platforms/apple/AppleCodec.cpp)
    set(library_type STATIC)
elseif(WIN32)
    list(APPEND core_sources platforms/windows/WindowsCodec.cpp platforms/windows/wic/ImageProcessor.cpp)
    if(BUILD_SHARED_LIBS)
        set(library_type SHARED)
    else()
        set(library_type STATIC)
    endif()
else()
    message(FATAL_ERROR "Supported targets are Windows, iOS, macOS and Android.")
endif()

set(PH_IMAGE_NOTICES "")
foreach(notice COPYING PATENTS AUTHORS)
    file(READ "${webp_root}/${notice}" content)
    string(APPEND PH_IMAGE_NOTICES "${notice}\n${content}\n")
endforeach()
configure_file(cmake/third_party_notices.hpp.in generated/third_party_notices.hpp @ONLY)

# 将第三方对象直接并入目标，Apple 的 .a 不要求调用方再链接第二个 libwebp 静态库。
add_library(PasteHubImage ${library_type} ${core_sources} $<TARGET_OBJECTS:ph_webp>)
set(image_targets PasteHubImage)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_library(PasteHubImageShared SHARED ${core_sources} $<TARGET_OBJECTS:ph_webp>)
    set_target_properties(PasteHubImageShared PROPERTIES OUTPUT_NAME PasteHubImage INSTALL_NAME_DIR "@rpath")
    target_compile_definitions(PasteHubImageShared PUBLIC PH_IMAGE_SHARED)
    target_link_options(PasteHubImageShared PRIVATE "LINKER:-dead_strip"
        "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/platforms/apple/exports.txt")
    set_property(TARGET PasteHubImageShared APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/platforms/apple/exports.txt")
    list(APPEND image_targets PasteHubImageShared)
endif()

foreach(image_target IN LISTS image_targets)
    target_include_directories(${image_target} PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
    target_include_directories(${image_target} PRIVATE src "${webp_root}/src" "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(${image_target} PRIVATE PH_IMAGE_BUILD)
    if(library_type STREQUAL "SHARED")
        target_compile_definitions(${image_target} PUBLIC PH_IMAGE_SHARED)
    endif()

    if(APPLE)
        target_link_libraries(${image_target} PUBLIC "-framework CoreFoundation" "-framework CoreGraphics" "-framework ImageIO")
    elseif(ANDROID)
        set_target_properties(${image_target} PROPERTIES OUTPUT_NAME pastehub_image)
        target_link_libraries(${image_target} PRIVATE jnigraphics log)
        target_compile_options(${image_target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-Os>" "$<$<NOT:$<CONFIG:Debug>>:-g0>")
        target_link_options(${image_target} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--gc-sections"
            "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/platforms/android/exports.map"
            "$<$<NOT:$<CONFIG:Debug>>:-Wl,--strip-all>")
        set_property(TARGET ${image_target} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/platforms/android/exports.map")
    elseif(WIN32)
        target_compile_definitions(${image_target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        target_link_libraries(${image_target} PUBLIC windowscodecs ole32 uuid)
    endif()

    if(MSVC)
        target_compile_options(${image_target} PRIVATE /W4 /utf-8 /Gy /Gw)
    else()
        target_compile_options(${image_target} PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
    endif()
endforeach()
if(ANDROID)
    target_compile_options(ph_webp PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-g0>")
endif()
if(NOT MSVC)
    target_compile_options(ph_webp PRIVATE -Os -ffunction-sections -fdata-sections)
endif()

install(TARGETS ${image_targets} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/pastehub_image.h DESTINATION include)
install(FILES vendor/libwebp/COPYING vendor/libwebp/PATENTS vendor/libwebp/AUTHORS DESTINATION share/licenses/libwebp)
if(APPLE)
    install(FILES platforms/apple/module.modulemap DESTINATION include)
endif()
